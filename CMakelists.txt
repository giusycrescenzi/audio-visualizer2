cmake_minimum_required(VERSION 3.16)
project(MusicArt2 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Variabili per i path (più leggibile e manutenibile)
set(QT_PREFIX /opt/homebrew/opt/qt)
set(SFML_PREFIX /opt/homebrew/opt/sfml)
set(FFTW_PREFIX /opt/homebrew/opt/fftw)

#non so cosa sia ma serve
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)

# Includi i path manualmente (per FFTW e sfml)include_directories(${Qt6Widgets_INCLUDE_DIRS})
include_directories(${FFTW_PREFIX}/include)
link_directories(${FFTW_PREFIX}/lib)
include_directories(${SFML_PREFIX}/include)
link_directories(${SFML_PREFIX}/lib)

# Aiuta CMake a trovare SFML (tramite il suo pacchetto CMake)
set(SFML_DIR ${SFML_PREFIX}/lib/cmake/SFML)

# Trova i moduli Qt6
find_package(Qt6 REQUIRED Core Gui Widgets)

# File sorgente
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS src/*.cpp)
file(GLOB_RECURSE HEADERS CONFIGURE_DEPENDS src/*.h)

# Crea eseguibile
add_executable(MusicArt2 ${SOURCES} ${HEADERS})

target_link_libraries(MusicArt2
    PRIVATE
        Qt6::Core
        Qt6::Gui
        Qt6::Widgets
        sfml-graphics
        sfml-window
        sfml-system
        sfml-audio
        fftw3
)

# Copia assets accanto all’eseguibile
add_custom_command(TARGET MusicArt2 POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory
        ${CMAKE_SOURCE_DIR}/assets $<TARGET_FILE_DIR:MusicArt2>/assets
)
